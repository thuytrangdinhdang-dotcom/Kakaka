local isRunning = false
local lockedCameraCFrame = nil

local function startAutoMaster()
    if isRunning then return end
    isRunning = true
    ToggleMasterBtn.Text = "🛑 TẮT AUTO (Đang chạy)"
    ToggleMasterBtn.BackgroundColor3 = Color3.fromRGB(220, 50, 50)
    StatusLabel.Text = "✨ Đang khóa camera cố định bên tay phải..."
    
    local lp = Players.LocalPlayer
    local camera = workspace.CurrentCamera
    
    if lp.Character and lp.Character:FindFirstChild("Humanoid") then
        if not isSpeedBuffEnabled then
            lp.Character.Humanoid.WalkSpeed = 0
        end
        lp.Character.Humanoid.JumpPower = 0
    end
    
    camera.CameraType = Enum.CameraType.Scriptable
    
    if lp.Character and lp.Character:FindFirstChild("Head") then
        local headPos = lp.Character.Head.Position
        local rightOffset = (lp.Character.Head.CFrame.RightVector * 5) + Vector3.new(0, 2, 0)
        local camTargetPos = headPos + rightOffset
        lockedCameraCFrame = CFrame.new(camTargetPos, headPos)
        camera.CFrame = lockedCameraCFrame
    end
    
    StatusLabel.Text = "🔥 Đang chạy Auto Câu + Skill..."
    
    task.spawn(function()
        local targetSkillNames = {
            ["Slot1"] = true,
            ["Slot2"] = true,
            ["Slot3"] = true,
            ["Slot4"] = true,
            ["SlotTemplate"] = true
        }
        
        while isRunning do
            if lockedCameraCFrame then
                workspace.CurrentCamera.CFrame = lockedCameraCFrame
            end
            
            pcall(function()
                for _, gui in ipairs(PlayerGui:GetChildren()) do
                    if gui:IsA("ScreenGui") and isRunning then
                        for _, desc in ipairs(gui:GetDescendants()) do
                            if desc.Name == "FishingActionButton" and desc:IsA("ImageButton") and desc.Visible and isRunning then
                                local absPos = desc.AbsolutePosition
                                local absSize = desc.AbsoluteSize
                                local baseCenterX = absPos.X + (absSize.X / 2)
                                local baseCenterY = absPos.Y + (absSize.Y / 2)
                                
                                local finalX = baseCenterX + sellOffsetX
                                local finalY = baseCenterY + sellOffsetY
                                
                                Indicator.Position = UDim2.new(0, finalX, 0, finalY)
                                Indicator.Visible = true
                                
                                VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, true, game, 0)
                                task.wait(0.6)
                                VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, false, game, 0)
                                task.wait(0.05)
                                
                                for i = 1, 3 do
                                    if not isRunning then break end
                                    VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, true, game, 0)
                                    task.wait(0.02)
                                    VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, false, game, 0)
                                    task.wait(0.05)
                                end

                                local vpSize = workspace.CurrentCamera.ViewportSize
                                local midX = (vpSize.X / 2) + sellOffsetX
                                local midY = (vpSize.Y / 2) + sellOffsetY
                                VirtualInputManager:SendMouseButtonEvent(midX, midY, 0, true, game, 0)
                                task.wait(0.02)
                                VirtualInputManager:SendMouseButtonEvent(midX, midY, 0, false, game, 0)
                                task.wait(0.05)
                            end
                        end
                    end
                end
            end)
            
            pcall(function()
                for _, gui in ipairs(PlayerGui:GetChildren()) do
                    if gui:IsA("ScreenGui") and isRunning then
                        for _, desc in ipairs(gui:GetDescendants()) do
                            if targetSkillNames[desc.Name] and desc.Visible and isRunning then
                                local absPos = desc.AbsolutePosition
                                local absSize = desc.AbsoluteSize
                                local baseCenterX = absPos.X + (absSize.X / 2)
                                local baseCenterY = absPos.Y + (absSize.Y / 2)
                                
                                local finalX = baseCenterX + sellOffsetX
                                local finalY = baseCenterY + sellOffsetY
                                
                                VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, true, game, 0)
                                task.wait(0.01)
                                VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, false, game, 0)
                                task.wait(0.01)
                            end
                        end
                    end
                end
            end)
            
            task.wait(0.01)
        end
    end)
end

-- ================= HÀM BAY XUYÊN ĐẤT, NOCLIP & TRỒI LÊN CHUẨN XÁC =================
local function flySmoothlyTo(targetCFrame)
    local lp = Players.LocalPlayer
    local char = lp.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") or not char:FindFirstChild("Humanoid") then return false end
    
    local hrp = char.HumanoidRootPart
    local humanoid = char.Humanoid
    
    humanoid.PlatformStand = true
    local speed = 70
    local runServiceConnection
    local noclipConnection
    
    noclipConnection = RunService.Stepped:Connect(function()
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end
    end)
    
    runServiceConnection = RunService.Stepped:Connect(function(_, dt)
        if not char or not char:FindFirstChild("HumanoidRootPart") then
            if runServiceConnection then runServiceConnection:Disconnect() end
            if noclipConnection then noclipConnection:Disconnect() end
            return
        end
        
        local currentPos = hrp.Position
        local undergroundTarget = targetCFrame.Position - Vector3.new(0, 25, 0)
        local direction = (undergroundTarget - currentPos)
        local distance = direction.Magnitude
        local surfaceDistance = (Vector3.new(currentPos.X, targetCFrame.Position.Y, currentPos.Z) - targetCFrame.Position).Magnitude
        
        if surfaceDistance < 6 then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            if runServiceConnection then runServiceConnection:Disconnect() end
        else
            local velocity = direction.Unit * math.min(speed, distance * 10)
            hrp.AssemblyLinearVelocity = velocity
            hrp.CFrame = CFrame.new(currentPos, undergroundTarget)
        end
    end)
    
    local timeout = tick() + 15
    while (Vector3.new(hrp.Position.X, targetCFrame.Position.Y, hrp.Position.Z) - targetCFrame.Position).Magnitude > 6 and tick() < timeout and char:FindFirstChild("HumanoidRootPart") do
        task.wait(0.05)
    end
    
    if runServiceConnection then runServiceConnection:Disconnect() end
    if noclipConnection then noclipConnection:Disconnect() end
    
    if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("Humanoid") then
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        local surfacePos = targetCFrame + Vector3.new(0, 6, 0)
        hrp.CFrame = surfacePos
        task.wait(0.15)
        
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = true
            end
        end
        
        humanoid.PlatformStand = false
        hrp.CFrame = targetCFrame
    end
    
    local finalDist = (hrp.Position - targetCFrame.Position).Magnitude
    return finalDist <= 12
end

                    local shortestDist = math.huge
                    
                    for _, obj in ipairs(workspace:GetDescendants()) do
                        local nameLower = string.lower(obj.Name)
                        if nameLower == "npc_fish_seller_1" or string.find(nameLower, "seller") or string.find(nameLower, "fish_seller") then
                            local part = nil
                            if obj:IsA("Model") then
                                part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                            elseif obj:IsA("BasePart") then
                                part = obj
                            end
                            
                            if part then
                                local dist = (char.HumanoidRootPart.Position - part.Position).Magnitude
                                if dist < shortestDist then
                                    shortestDist = dist
                                    targetNpcPart = part
                                end
                            end
                        end
                    end
                    
                    if targetNpcPart then
                        StatusLabel.Text = "🕳️ Đang bay xuyên lòng đất tới NPC..."
                        local targetPos = targetNpcPart.Position + Vector3.new(0, 3, 5)
                        
                        local arrivedAtNpc = flySmoothlyTo(CFrame.new(targetPos))
                        
                        if arrivedAtNpc then
                            StatusLabel.Text = "✅ Đã tới vị trí bán cá!"
                            task.wait(0.5)
                            
                            pcall(function()
                                for _, obj in ipairs(workspace:GetDescendants()) do
                                    local nameLower = string.lower(obj.Name)
                                    if nameLower == "npc_fish_seller_1" or string.find(nameLower, "seller") then
                                        local part = nil
                                        if obj:IsA("Model") then
                                            part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                                        elseif obj:IsA("BasePart") then
                                            part = obj
                                        end
                                        
                                        if part and (part.Position - char.HumanoidRootPart.Position).Magnitude < 30 then
                                            for _, prompt in ipairs(part:GetDescendants()) do
                                                if prompt:IsA("ProximityPrompt") then
                                                    fireproximityprompt(prompt)
                                                end
                                            end
                                        end
                                    end
                                end
                            end)
                            
                            StatusLabel.Text = "⏳ Đợi sau khi nhấn talk (2s)..."
                            task.wait(2.0)
                            
                            local function clickButtonByName(btnName)
                                local clicked = false
                                pcall(function()
                                    for _, gui in ipairs(PlayerGui:GetChildren()) do
                                        if gui:IsA("ScreenGui") then
                                            for _, desc in ipairs(gui:GetDescendants()) do
                                                if desc.Name == btnName and desc:IsA("TextButton") and desc.Visible then
                                                    local absPos = desc.AbsolutePosition
                                                    local absSize = desc.AbsoluteSize
                                                    local baseCenterX = absPos.X + (absSize.X / 2)
                                                    local baseCenterY = absPos.Y + (absSize.Y / 2)
                                                    
                                                    local finalX = baseCenterX + sellOffsetX
                                                    local finalY = baseCenterY + sellOffsetY
                                                    
                                                    VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, true, game, 0)
                                                    task.wait(0.05)
                                                    VirtualInputManager:SendMouseButtonEvent(finalX, finalY, 0, false, game, 0)
                                                    clicked = true
                                                end
                                            end
                                        end
                                    end
                                end)
                                return clicked
                            end
                            
                            StatusLabel.Text = "🛒 Nhấn Option002 lần 1..."
                            clickButtonByName("Option002")
                            
                            StatusLabel.Text = "⏳ Đợi 2s trước khi nhấn Option001..."
                            task.wait(2.0)
                            
                            StatusLabel.Text = "🛒 Nhấn Option001..."
                            clickButtonByName("Option001")
                            
                            StatusLabel.Text = "⏳ Đợi 2s trước khi nhấn Option002 lần 2..."
                            task.wait(2.0)
                            
                            StatusLabel.Text = "🛒 Nhấn Option002 lần 2..."
                            clickButtonByName("Option002")
                            
                            task.wait(1.0)
                        else
                            StatusLabel.Text = "⚠️ Không thể bay tới vị trí NPC!"
                            task.wait(2)
                        end
                        
                        StatusLabel.Text = "🕳️ Đang bay xuyên lòng đất về vị trí cũ..."
                        local arrivedAtOldPos = flySmoothlyTo(originalCFrame)
                        
                        if arrivedAtOldPos then
                            StatusLabel.Text = "✅ Đã về lại vị trí cũ chính xác!"
                        else
                            if char and char:FindFirstChild("HumanoidRootPart") then
                                char.HumanoidRootPart.CFrame = originalCFrame
                            end
                            StatusLabel.Text = "✨ Đã khôi phục vị trí câu cũ."
                        end
                        
                    else
                        StatusLabel.Text = "⚠️ Không tìm thấy NPC bán cá!"
                        task.wait(2)
                    end
                end
                
                if isAutoSellRunning then
                    StatusLabel.Text = "🚀 Đang tự động bật lại auto câu..."
                    task.wait(1)
                    if isAutoSellRunning then
                        startAutoMaster()
                    end
                end
                
                task.wait(1)
            end
        end)
    else
        AutoSellBtn.Text = "📦 Auto Sell: TẮT"
        AutoSellBtn.BackgroundColor3 = Color3.fromRGB(180, 100, 40)
        StatusLabel.Text = "⚡ Trạng thái: Đang tắt"
    end
end)
